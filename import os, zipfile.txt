    "server.js": """const http = require("node:http");
const { URL } = require("node:url");

const PORT = Number(process.env.PORT || 3000);
const ALLOWED_ORIGIN = process.env.ALLOWED_ORIGIN || "*";
const MAX_BODY_BYTES = 16 * 1024;

function send(res, status, payload) {
  const body = JSON.stringify(payload);
  res.writeHead(status, {
    "Content-Type": "application/json; charset=utf-8",
    "Content-Length": Buffer.byteLength(body),
    "Cache-Control": "no-store"
  });
  res.end(body);
}

function setCors(req, res) {
  const origin = req.headers.origin;
  if (ALLOWED_ORIGIN === "*" || origin === ALLOWED_ORIGIN) {
    res.setHeader("Access-Control-Allow-Origin", ALLOWED_ORIGIN === "*" ? "*" : origin);
  }
  res.setHeader("Vary", "Origin");
  res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Content-Type, Authorization");
}

function readJson(req) {
  return new Promise((resolve, reject) => {
    let raw = "";
    let size = 0;
    req.on("data", chunk => {
      size += chunk.length;
      if (size > MAX_BODY_BYTES) {
        reject(Object.assign(new Error("Request body too large"), { status: 413 }));
        req.destroy();
        return;
      }
      raw += chunk;
    });
    req.on("end", () => {
      try { resolve(raw ? JSON.parse(raw) : {}); }
      catch { reject(Object.assign(new Error("Request body must be valid JSON"), { status: 400 })); }
    });
    req.on("error", reject);
  });
}

const server = http.createServer(async (req, res) => {
  setCors(req, res);
  if (req.method === "OPTIONS") {
    res.writeHead(204);
    return res.end();
  }

  const url = new URL(req.url, `http://${req.headers.host || "localhost"}`);

  if (req.method === "GET" && url.pathname === "/") {
    return send(res, 200, {
      name: "Fortnite Replay Viewer API",
      status: "online",
      endpoints: ["/health", "/api/replays/import"]
    });
  }

  if (req.method === "GET" && url.pathname === "/health") {
    return send(res, 200, { ok: true, service: "replay-viewer-backend" });
  }

  if (req.method === "POST" && url.pathname === "/api/replays/import") {
    let data;
    try { data = await readJson(req); }
    catch (err) { return send(res, err.status || 400, { error: err.message || "Could not read request" }); }

    const code = String(data.code || "").trim();
    if (!/^[a-fA-F0-9]{32}$/.test(code)) {
      return send(res, 400, { error: "Replay code must be exactly 32 hexadecimal characters." });
    }

    // Placeholder: a supported replay provider/parser must be connected here.
    return send(res, 501, {
      error: "Replay import is not configured yet.",
      detail: "The API is online, but no replay provider/parser is connected."
    });
  }

  return send(res, 404, { error: "Endpoint not found" });
});

server.listen(PORT, "0.0.0.0", () => {
  console.log(`Replay viewer backend listening on port ${PORT}`);
